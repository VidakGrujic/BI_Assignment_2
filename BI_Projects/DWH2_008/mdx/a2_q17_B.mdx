WITH
  MEMBER [Measures].[Avg Data Quality 2023] AS
    ([dim_timemonth].[Time].[year_num].&[2023],
     [Measures].[Avg Data Quality])

  MEMBER [Measures].[Avg Data Quality 2024] AS
    ([dim_timemonth].[Time].[year_num].&[2024],
     [Measures].[Avg Data Quality])

SELECT
  {
    [Measures].[Avg Data Quality 2023],
    [Measures].[Avg Data Quality 2024]
  } ON COLUMNS,

  NONEMPTY(
    [dim_city].[Geo].[country_name].Members,
    [Measures].[Avg Data Quality]
  ) ON ROWS

FROM [AirQ Cube]
